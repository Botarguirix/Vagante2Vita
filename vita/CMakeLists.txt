cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
endif()
project(Vagante2VitaDiagnostic C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
add_executable(vagante2vita src/main.c)
target_compile_options(vagante2vita PRIVATE -Wall -Wextra -Werror)
target_link_libraries(vagante2vita SceDisplay_stub SceCtrl_stub SceSysmem_stub SceLibKernel_stub)
vita_create_self(eboot.bin vagante2vita UNSAFE)
vita_create_vpk(Vagante2Vita-diagnostic.vpk VAGV00001 eboot.bin VERSION 00.01 NAME "Vagante2Vita Diagnostic")
