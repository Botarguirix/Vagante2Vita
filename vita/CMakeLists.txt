cmake_minimum_required(VERSION 3.16)
option(HOST_PROBE "Build desktop archive/render probe" OFF)
if(NOT HOST_PROBE AND NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
endif()
project(Vagante2Vita C)
include("${CMAKE_CURRENT_SOURCE_DIR}/iteration.cmake")
add_library(miniz STATIC ../third_party/miniz/miniz.c ../third_party/miniz/miniz_tinfl.c)
target_include_directories(miniz PUBLIC ../third_party/miniz)
target_compile_definitions(miniz PUBLIC MINIZ_NO_ARCHIVE_APIS MINIZ_NO_DEFLATE_APIS)
add_library(resources STATIC src/resources.c src/scene.c src/menu.c src/world.c src/player.c src/world_visuals.c src/world_entities.c)
target_include_directories(resources PRIVATE ../third_party/stb)
target_link_libraries(resources PUBLIC miniz m)
if(HOST_PROBE)
  target_compile_definitions(resources PUBLIC HOST_PROBE=1)
  add_executable(vagante-probe src/probe.c)
  target_link_libraries(vagante-probe resources)
  enable_testing()
  add_executable(frame-upload-test ../tests/frame_upload_test.c)
  target_include_directories(frame-upload-test PRIVATE src)
  target_compile_options(frame-upload-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME dirty-frame-upload COMMAND frame-upload-test)
  add_executable(menu-test ../tests/menu_test.c src/menu.c)
  target_include_directories(menu-test PRIVATE src)
  target_compile_options(menu-test PRIVATE -UNDEBUG)
  add_test(NAME menu-transitions COMMAND menu-test)
  add_executable(session-log-test ../tests/session_log_test.c src/session_log.c)
  target_include_directories(session-log-test PRIVATE src)
  target_compile_options(session-log-test PRIVATE -UNDEBUG)
  add_test(NAME session-log-preservation COMMAND session-log-test)
  add_executable(world-entities-test ../tests/world_entities_test.c src/world_entities.c)
  target_include_directories(world-entities-test PRIVATE src)
  target_compile_options(world-entities-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME world-entity-packet COMMAND world-entities-test)
  add_executable(world-runtime-probe src/world_runtime_probe.c)
  target_link_libraries(world-runtime-probe resources)
  add_executable(player-test ../tests/player_test.c src/player.c src/world.c)
  target_include_directories(player-test PRIVATE src)
  target_compile_options(player-test PRIVATE -UNDEBUG)
  target_link_libraries(player-test m)
  add_test(NAME player-movement COMMAND player-test)
  add_executable(world-probe src/world_probe.c src/world.c)
  add_executable(world-test ../tests/world_test.c src/world.c)
  target_include_directories(world-test PRIVATE src)
  target_compile_options(world-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME world-flags-and-stats COMMAND world-test)
  return()
endif()
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
add_executable(vagante2vita src/main.c src/session_log.c)
target_compile_options(vagante2vita PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(vagante2vita PRIVATE
  VAGANTE_ITERATION="${VAGANTE_ITERATION}" VAGANTE_VERSION="${VITA_VERSION}")
target_link_libraries(vagante2vita resources SceDisplay_stub SceCtrl_stub SceSysmem_stub SceLibKernel_stub)
vita_create_self(eboot.bin vagante2vita UNSAFE)
vita_create_vpk(Vagante2Vita-iteration${VAGANTE_ITERATION}.vpk ${VITA_TITLEID} eboot.bin VERSION ${VITA_VERSION} NAME "${VITA_APP_NAME}"
  FILE sce_sys/icon0.png sce_sys/icon0.png
  FILE sce_sys/pic0.png sce_sys/pic0.png
  FILE sce_sys/livearea/contents/bg0.png sce_sys/livearea/contents/bg0.png
  FILE sce_sys/livearea/contents/startup.png sce_sys/livearea/contents/startup.png
  FILE sce_sys/livearea/contents/template.xml sce_sys/livearea/contents/template.xml)
